} return ss; } function doGet(e) { try { const action = e.parameter.action; let result; if (action === 'getLog') result = getLog(); else if (action === 'getUsers') result = getUsers(); else if (action === 'login') result = login(e); else result = {error: 'Unbekannte Aktion'}; return ContentService.createTextOutput(JSON.stringify(result)).setMimeType(ContentService.MimeType.JSON); } catch(err) { return ContentService.createTextOutput(JSON.stringify({error: err.toString()})).setMimeType(ContentService.MimeType.JSON); } } function doPost(e) { try { const data = JSON.parse(e.postData.contents); const action = data.action; let result; if (action === 'addLog') result = addLog(data); else if (action === 'updateLog') result = updateLog(data); else if (action === 'deleteLog') result = deleteLog(data); else if (action === 'addUser') result = addUser(data); else if (action === 'deleteUser') result = deleteUser(data); else if (action === 'deleteDateRange') result = deleteDateRange(data); else result = {error: 'Unbekannte Aktion'}; return ContentService.createTextOutput(JSON.stringify(result)).setMimeType(ContentService.MimeType.JSON); } catch(err) { return ContentService.createTextOutput(JSON.stringify({error: err.toString()})).setMimeType(ContentService.MimeType.JSON); } } function getLog() { const ss = SpreadsheetApp.openById(SHEET_ID); const sheet = ss.getSheetByName(LOG_SHEET); if (!sheet || sheet.getLastRow() < 2) return {log: []}; const data = sheet.getRange(2, 1, sheet.getLastRow()-1, 9).getValues(); const log = data .map((row, i) => ({ rowIndex: i + 2, ts: String(row[0]), // Always return as string fili: String(row[1]), dest: String(row[2]), order: String(row[3]), artno: String(row[4]), groesse: String(row[5]), format: String(row[6]), user: String(row[7]), rawBarcode: String(row[8]) })) .filter(e => e.ts && e.ts !== '' && e.fili !== '') .reverse(); // Write timestamp as plain text to prevent Google Sheets date conversion sheet.getRange(lastRow, 1).setNumberFormat('@').setValue(data.ts); sheet.getRange(lastRow, 2, 1, 8).setValues([[ data.fili, data.dest, data.order, data.artno, data.groesse, data.format || '', data.user || '–', data.rawBarcode || '' ]]); return {success: true}; } function deleteLog(data) { const ss = SpreadsheetApp.openById(SHEET_ID); const sheet = ss.getSheetByName(LOG_SHEET); if (data.rowIndex && data.rowIndex > 1) sheet.deleteRow(data.rowIndex); return {success: true}; } function deleteDateRange(data) { const ss = SpreadsheetApp.openById(SHEET_ID); const sheet = ss.getSheetByName(LOG_SHEET); if (!sheet || sheet.getLastRow() < 2) return {success: true, deleted: 0}; const from = data.from ? new Date(data.from) : null; const to = data.to ? new Date(data.to) : null; if (to) to.setHours(23,59,59); const rows = sheet.getRange(2, 1, sheet.getLastRow()-1, 1).getValues(); const toDelete = []; rows.forEach((row, i) => { const d = new Date(row[0]); if (from && d < from) return; if (to && d > to) return; toDelete.push(i + 2); }); toDelete.reverse().forEach(r => sheet.deleteRow(r)); return {success: true, deleted: toDelete.length}; } function getUsers() { const ss = SpreadsheetApp.openById(SHEET_ID); const sheet = ss.getSheetByName(USER_SHEET); if (!sheet || sheet.getLastRow() < 2) return {users: []}; const data = sheet.getRange(2, 1, sheet.getLastRow()-1, 4).getValues(); const users = data.map((row, i) => ({ rowIndex: i + 2, id: String(row[0]), name: String(row[1]), password: String(row[2]), role: String(row[3]) })).filter(u => u.id && u.id !== ''); return {users}; } function login(e) { const userId = e.parameter.userId; const password = e.parameter.password; const result = getUsers(); const user = result.users.find(u => u.id === userId && u.password === password); const sheet = ss.getSheetByName(USER_SHEET); const existing = getUsers(); if (existing.users.find(u => u.id === data.id)) return {error: 'Login-ID bereits vergeben'}; sheet.appendRow([data.id, data.name, data.password, data.role || 'admin']); if (data.rowIndex && data.rowIndex > 1) sheet.deleteRow(data.rowIndex); return {success: true}; } function setup() { initSheets(); // Set column A to text format const ss = SpreadsheetApp.openById(SHEET_ID);
